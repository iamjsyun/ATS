    public class BuyOption
    {
        public bool Enable { get; set; } = true;
        public int GridCount { get; set; } = 5;
        public List<string> Profiles { get; set; } = new();

        public List<XGridProfile> GetProfiles(int cno)
        {
            var result = new List<XGridProfile>();
            if (Profiles == null) return result;

            foreach (var str in Profiles)
            {
                if (string.IsNullOrWhiteSpace(str)) continue;
                var parts = str.Split(',');
                // 그리드 프로필 (v3.3): {GNO},{OrderType},{LotType},{Lot},{limit offset},{te start},{te step},{ts start},{ts step},{tp},{sl} (총 11개 필드)
                if (parts.Length < 11) continue;

                try
                {
                    result.Add(new XGridProfile
                    {
                        cno = cno,
                        dir = 1,
                        gno = int.Parse(parts[0].Trim()),
                        type = int.Parse(parts[1].Trim()),
                        lot_type = int.Parse(parts[2].Trim()),
                        lot = double.Parse(parts[3].Trim()),
                        offset = double.Parse(parts[4].Trim()),
                        te_start = double.Parse(parts[5].Trim()),
                        te_step = double.Parse(parts[6].Trim()),
                        ts_start = int.Parse(parts[7].Trim()),
                        ts_step = int.Parse(parts[8].Trim()),
                        tp = double.Parse(parts[9].Trim()),
                        sl = double.Parse(parts[10].Trim()),
                        gap_min = 200
                    });
                }
                catch { /* skip invalid format */ }
            }
            return result;
        }
    }
