/**
 * @class CXOrderManager
 * @brief 샌드박스 세션 내의 주문 실행 및 관리 담당
 */
class CXOrderManager : public IXOrderManager {
private:
    ulong           m_ticket;
    string          m_sid;
    CTrade          m_trade;
    ICXContext*     m_ctx;

public:
    CXOrderManager(ICXContext* ctx) : m_ctx(ctx), m_ticket(0), m_sid("") {}
    virtual ~CXOrderManager() override {}

    virtual void SetMagic(ulong magic) override { m_trade.SetExpertMagicNumber(magic); }

    /**
     * @brief 진입 주문 실행 (xa_entry 기반)
     */
    virtual bool ExecuteEntry(ICXParam* xp) override {
        if(IS_NULL(m_ctx) || IS_NULL(xp)) return false;

        ICXSignal* sig = xp.GetSignal();
        if(IS_INVALID(sig)) return false;

        // [Removed redundant CXGuard validation]
        // SignalWatcher already performs strict validation before binding the signal to the session.
        // Failing here causes an immediate SESSION_ERROR.

        m_sid = sig.GetSid();
        m_trade.SetExpertMagicNumber((ulong)sig.GetMagic());

        XP_LOG_INFO(xp, StringFormat("[EXEC-ENTRY] SID:%s, Dir:%d, Type:%d, Lot:%.2f, Price:%.5f", 
           m_sid, sig.GetDir(), sig.GetType(), sig.GetLot(), sig.GetPriceSignal()));

        //--- 주문 타입 및 방향 결정 (DIP/Unified)
        ENUM_ORDER_TYPE order_type;
        // [Fix] Type 1 (ORDER_LIMIT_TRAILING)은 대기 주문(Pending)으로 처리
        bool isMarket = sig.IsMarketOrder();
        
        if(isMarket) {
            order_type = (sig.GetDir() == CX_DIR_BUY) ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
        } else {
            // Type 1을 포함한 그 외는 모두 대기 주문으로 처리
            order_type = (sig.GetDir() == CX_DIR_BUY) ? ORDER_TYPE_BUY_LIMIT : ORDER_TYPE_SELL_LIMIT;
        }

        //--- 단일 OrderSend로 통합 실행 (if-else 최소화)
        string sym = sig.GetSymbol();
        if(!SymbolSelect(sym, true)) {
            string err_msg = StringFormat("SymbolSelect failed for %s. Cannot trade.", sym);
            XP_LOG_ERROR(xp, err_msg);
            CXMessageProvider::UpdateStatus(sig, XE_ERROR, err_msg);
            return false;
        }

        double price = sig.GetPriceSignal();
        // [Fix] 시장가/지정가 모두 더미 가격(1.0 이하)인 경우 현재 호가로 보정
        if(price <= 1.0) {
            price = (sig.GetDir() == CX_DIR_BUY) ? SymbolInfoDouble(sym, SYMBOL_ASK) : SymbolInfoDouble(sym, SYMBOL_BID);
        }

        //--- [Fix] SL/TP 값을 포인트(거리)에서 절대 가격(Absolute Price)으로 변환 (IXGuard 활용)
        double sl_price = 0.0;
        double tp_price = 0.0;
        IXGuard* guard = CX_GET_OBJ(m_ctx, "guard", IXGuard);

        if(IS_VALID(guard)) {
            if(sig.GetSL() > 0) {
                double sl_dist = guard.PointsToPrice(sym, sig.GetSL());
                sl_price = (sig.GetDir() == CX_DIR_BUY) ? (price - sl_dist) : (price + sl_dist);
                sl_price = guard.NormalizePrice(sym, sl_price);
            }

            if(sig.GetTP() > 0) {
                double tp_dist = guard.PointsToPrice(sym, sig.GetTP());
                tp_price = (sig.GetDir() == CX_DIR_BUY) ? (price + tp_dist) : (price - tp_dist);
                tp_price = guard.NormalizePrice(sym, tp_price);
            }

            //-- [Defensive] StopsLevel 위반 시 안전하게 SL/TP 초기화
            if(sl_price > 0 && !guard.ValidateStopLevel(sym, price, sl_price)) {
                XP_LOG_WARN(xp, StringFormat("[EXEC-ENTRY] SL too close (StopsLevel), resetting SL to 0.0. SID:%s", m_sid));
                sl_price = 0.0;
            }
            if(tp_price > 0 && !guard.ValidateStopLevel(sym, price, tp_price)) {
                XP_LOG_WARN(xp, StringFormat("[EXEC-ENTRY] TP too close (StopsLevel), resetting TP to 0.0. SID:%s", m_sid));
                tp_price = 0.0;
            }
        }
        
        XP_LOG_DEBUG(xp, StringFormat("[EXEC-DEBUG] OrderOpen -> Sym:%s, Type:%d, Lot:%.2f, Price:%.5f, SL:%.5f, TP:%.5f", 
            sym, order_type, sig.GetLot(), price, sl_price, tp_price));

        //--- [Fix] 시장가 주문 시 CTrade::Buy/Sell 명시적 호출로 10013 에러 회피
        bool res = false;
        if(isMarket) {
            if(sig.GetDir() == CX_DIR_BUY) 
                res = m_trade.Buy(sig.GetLot(), sym, price, sl_price, tp_price, m_sid);
            else 
                res = m_trade.Sell(sig.GetLot(), sym, price, sl_price, tp_price, m_sid);
        } else {
            res = m_trade.OrderOpen(sym, order_type, sig.GetLot(), 0, price, sl_price, tp_price, ORDER_TIME_GTC, 0, m_sid);
        }

        if(!res) {
            string err_msg = StringFormat("%s (Retcode: %d - %s)", 
                CXMessageProvider::Format(MSG_ERR_TRADE_FAILED, ""), 
                m_trade.ResultRetcode(), 
                m_trade.ResultRetcodeDescription());
            XP_LOG_ERROR(xp, err_msg);
            CXMessageProvider::UpdateStatus(sig, XE_ERROR, err_msg);
            return false;
        }
        else {
            m_ticket = m_trade.ResultOrder();
            
            //-- [Fix] MT5 PositionSelectByTicket으로 검증 (Legacy OrderSelect 대신 사용)
            if(PositionSelectByTicket(m_ticket)) {
                sig.SetTicket((long)m_ticket);

                int next_status = (sig.GetType() == ORDER_MARKET) ? XE_EXECUTED : XE_PENDING_PLACED;
                string status_msg = (sig.GetType() == ORDER_MARKET) ? MSG_ENTRY_MARKET_SUCCESS : MSG_ENTRY_LIMIT_PLACED;

                CXMessageProvider::UpdateStatus(sig, next_status, status_msg);
                XP_LOG_OK(xp, status_msg);
                return true;
            } else {
                XP_LOG_ERROR(xp, StringFormat("[ORDER-MANAGER] Failed to confirm PositionSelectByTicket for ticket:%lld", m_ticket));
                CXMessageProvider::UpdateStatus(sig, XE_ERROR, "Order confirmation failed (PositionSelect)");
                return false;
            }
        }
    }

    /**
     * @brief 주문 수정 (진트레)
     */
    virtual bool ModifyOrder(ulong ticket, double price, double sl, double tp) override {
        if(ticket == 0) return false;
        return m_trade.OrderModify(ticket, price, sl, tp, ORDER_TIME_GTC, 0);
    }

    virtual bool DeleteOrder(ulong ticket) override {
        if(ticket == 0) return false;
        
        // 1. 삭제 시도
        if(!m_trade.OrderDelete(ticket)) return false;
        
        // 2. 삭제 확인: 터미널에서 선택되지 않아야 함
        // OrderSelect는 특정 티켓 주문을 선택하는 함수, 실패해야 삭제 완료임
        if(OrderSelect(ticket)) {
            XP_LOG_ERROR(NULL, StringFormat("[ORDER-MANAGER] Failed to confirm deletion for ticket:%lld", ticket));
            return false;
        }
        
        return true;
    }
};
