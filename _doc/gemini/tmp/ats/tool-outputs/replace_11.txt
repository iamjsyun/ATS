      string sql = StringFormat("SELECT * FROM signals WHERE sid='%s'", sid);

      if (LoadBySql(sql, GetPointer(list)) > 0) {

      }

   }

private:
   int LoadBySql(string sql, CArrayObj* list) {
      int handle = m_db.GetHandle();
      int req = DatabasePrepare(handle, sql);

      if (req == INVALID_HANDLE) {
         PrintFormat("DatabasePrepare failed: %d", GetLastError());
         return 0;
      }

      int count = 0;
      while (DatabaseRead(req)) {
         CXSignal* sig = new CXSignal();

         // 3. 컬럼 명칭을 인덱스로 변환하여 대입 (프로젝트 표준: 시각적 가독성 및 스키마 변경 대응)

         DatabaseColumnText(req,    GetColumnIndex(req, "sid"),          sig.sid);
         DatabaseColumnText(req,    GetColumnIndex(req, "gid"),          sig.gid);
         DatabaseColumnText(req,    GetColumnIndex(req, "symbol"),       sig.symbol);
         DatabaseColumnInteger(req, GetColumnIndex(req, "dir"),          sig.dir);
         DatabaseColumnInteger(req, GetColumnIndex(req, "type"),         sig.type);
         DatabaseColumnDouble(req,  GetColumnIndex(req, "lot"),          sig.lot);
         DatabaseColumnDouble(req,  GetColumnIndex(req, "price_signal"), sig.price_signal);
         DatabaseColumnDouble(req,  GetColumnIndex(req, "tp"),           sig.tp);
         DatabaseColumnDouble(req,  GetColumnIndex(req, "sl"),           sig.sl);
         DatabaseColumnInteger(req, GetColumnIndex(req, "te_start"),     sig.te_start);
         DatabaseColumnInteger(req, GetColumnIndex(req, "te_step"),      sig.te_step);
         DatabaseColumnInteger(req, GetColumnIndex(req, "ts_start"),     sig.ts_start);
         DatabaseColumnInteger(req, GetColumnIndex(req, "ts_step"),      sig.ts_step);
         DatabaseColumnInteger(req, GetColumnIndex(req, "xa_entry"),     sig.xa_entry);
         DatabaseColumnInteger(req, GetColumnIndex(req, "xa_exit"),      sig.xa_exit);
         DatabaseColumnInteger(req, GetColumnIndex(req, "xe_status"),    sig.xe_status);

         count++;
      }

      return count;
   }
